% \setbottommargin subtracts \footskip but the text area is positioned by
% \headheight+\headsep, so with no header/footer the text can run ~0.3cm lower
% than asked. Set the height directly so the bottom margin really is 2.0cm.
\setlength{\textheight}{\dimexpr\smpageheight-2.0cm-2.0cm\relax}

% Layout widths. The section rules and headings span the full \textwidth. Body
% content (intro, skills, bullets, dates) is inset by \tabcolsep on both sides,
% so left and right edges are symmetric and scale with the margins above.
\newlength{\contentwidth}
\AtBeginDocument{\setlength{\contentwidth}{\dimexpr\textwidth-2\tabcolsep\relax}}

% Print a nice blue section heading and full-width line.
\newcommand{\blocktitle}[1]{
\parbox{\textwidth}{
	\vspace{3mm}
	\noindent
	\textcolor{MidnightBlue}{
{\myfontd {#1}
	\vspace*{2mm}
	\hrule}
	\vspace*{4mm}
	\noindent
} } }


% Long headings and small right-side details (employment history).
\newenvironment{resumeblock}[1]{\blocktitle{#1}\par\nopagebreak}{}
% Plain unbreakable one-tabular entry (used for Education).
% Entry header/row layout: title at the left, dates flush right. The table is
% exactly as wide as the bullet lists below it (\contentwidth, indented by \tabcolsep),
% so the right-hand edges of dates and bullet text line up.
\newcommand{\resumetabbegin}{\begin{tabular*}{\dimexpr\textwidth-\tabcolsep\relax}{@{\hspace{\tabcolsep}}p{11.5cm}@{\extracolsep{\fill}}r@{}}}
\newcommand{\resumetabend}{\end{tabular*}}
\newenvironment{resumeentry}{\resumetabbegin}{\resumetabend\par}

% Employment entry: #1 holds the header rows (title and dates), set in a small
% tabular that is kept with the first bullet. The bullets that follow are a
% normal list, so a page break may fall between bullets, but never inside one
% (\interlinepenalty=10000 forbids breaking between lines of a bullet).
% leftmargin/rightmargin reproduce the old in-tabular-cell indentation.
\newenvironment{resumejob}[1]{%
  \noindent\resumetabbegin#1\resumetabend\par\nopagebreak
  \myfonta
  \begin{itemize}[leftmargin=31pt,labelindent=6pt,%
    rightmargin=\tabcolsep,%
    topsep=5.5pt,itemsep=0.3mm,parsep=1.0mm,beginpenalty=10000]%
  \interlinepenalty=10000
  % Lets TeX stretch spaces just enough to avoid a 1pt overhang on a tight line.
  \emergencystretch=1em
}{\end{itemize}}
