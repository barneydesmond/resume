% Typeface: Inter Light, with SemiBold for emphasis. Needs XeLaTeX or LuaLaTeX
% (fontspec); it will not build with pdflatex. The fonts are looked up by family
% name through fontconfig, so Inter's static OTFs must be installed as ordinary
% system fonts, e.g. in ~/.local/share/fonts (then run fc-cache -f).
\usepackage{fontspec}
\defaultfontfeatures{Ligatures=TeX}
\setmainfont{Inter Light}[BoldFont={Inter SemiBold},
  ItalicFont={Inter Light Italic},BoldItalicFont={Inter SemiBold Italic}]

% Inter has no real small caps (its smcp feature only covers 11 letters), so
% emulate them: lowercase letters become capitals at 80% size.
\makeatletter
\newcommand{\fakescsize}{\fontsize{\fpeval{0.8*\f@size}pt}{0pt}\selectfont}
\makeatother
\ExplSyntaxOn
\cs_new:Npn \fakesc_char:N #1
  {
    \token_if_letter:NTF #1
      {
        \int_compare:nTF { `#1 > 96 }
          { {\fakescsize\char_uppercase:N #1} }
          { #1 }
      }
      { #1 }
  }
\cs_new:Npn \fakesc #1
  {
    \tl_set:Nn \l_tmpa_tl {#1}
    \tl_replace_all:Nnn \l_tmpa_tl { ~ } { \space }
    \tl_map_function:NN \l_tmpa_tl \fakesc_char:N
  }
\ExplSyntaxOff
\renewcommand{\textsc}[1]{\fakesc{#1}}

% Plain unbreakable one-tabular entry (used for Education).
% Entry header/row layout: title at the left, dates flush right. The table is

% so the right-hand edges of dates and bullet text line up.
\newcommand{\resumetabbegin}{\begin{tabular*}{\dimexpr\textwidth-\tabcolsep\relax}{@{\hspace{\tabcolsep}}p{11.5cm}@{\extracolsep{\fill}}r@{}}}

\newenvironment{resumeentry}{\resumetabbegin}{\resumetabend\par}

% Employment entry: #1 holds the header rows (title and dates), set in a small
% tabular that is kept with the first bullet. The bullets that follow are a
% normal list, so a page break may fall between bullets, but never inside one
% (\interlinepenalty=10000 forbids breaking between lines of a bullet).
% itemsep/parsep are fractions of \baselineskip so they scale with \linespread.
% leftmargin/rightmargin reproduce the old in-tabular-cell indentation.
\newenvironment{resumejob}[1]{%
  \noindent\resumetabbegin#1\resumetabend\par\nopagebreak
  \fontbody
  \begin{itemize}[leftmargin=31pt,labelindent=6pt,%
    rightmargin=\tabcolsep,%
    topsep=5.5pt,itemsep=0.066\baselineskip,parsep=0.22\baselineskip,%
    beginpenalty=10000]%
  \interlinepenalty=10000
  \raggedright
}{\end{itemize}}
